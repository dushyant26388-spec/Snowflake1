-- SNOWFLAKE TIME TRAVEL

USE DATABASE TSESTDB;
USE SCHEMA TEST_SCHEMA;

-- 1. CREATE TABLE

CREATE OR REPLACE TABLE TIME_TRAVEL (
ID INT,
NAME VARCHAR(20),
SALARY NUMBER(10,2)
);

-- 2. INSERT DATA

INSERT INTO TIME_TRAVEL VALUES
(1, 'Rahul', 25000),
(2, 'Amit', 30000),
(3, 'Priya', 35000);

-- 3. CHECK ORIGINAL DATA

SELECT * FROM TIME_TRAVEL;

-- 4. UPDATE DATA

UPDATE TIME_TRAVEL
SET SALARY = 55000
WHERE ID = 1;

-- 5. CHECK UPDATED DATA

SELECT * FROM TIME_TRAVEL;

-- 6. TIME TRAVEL USING OFFSET

SELECT *
FROM TIME_TRAVEL
AT (OFFSET => -60);

-- 7. TIME TRAVEL USING TIMESTAMP

SELECT *
FROM TIME_TRAVEL
AT (
TIMESTAMP => '2026-11-15 02:09:00 -0800'::TIMESTAMP_TZ
);

-- 8. TIME TRAVEL USING STATEMENT

SELECT *
FROM TIME_TRAVEL
BEFORE (
STATEMENT => 'YOUR_QUERY_ID'
);
