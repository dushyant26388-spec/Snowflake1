----   -- Dynamic Data Masking using CONCAT() to partially hide email addresses based on user roles
use role accountadmin;

ALTER TABLE demo_db.public.customers
MODIFY COLUMN email
UNSET MASKING POLICY;

CREATE OR REPLACE MASKING POLICY demo_db.public.email_mask
AS (val STRING)
RETURNS STRING ->
    CASE
        WHEN CURRENT_ROLE() = 'ACCOUNTADMIN' THEN val
        ELSE CONCAT(LEFT(val, 1), '***', SUBSTR(val, POSITION('@' IN val)))
    END;


alter table demo_db.public.customers
modify column email
set masking policy demo_db.public.email_mask;


USE ROLE ANALYST;
SELECT * FROM demo_db.public.customers;
