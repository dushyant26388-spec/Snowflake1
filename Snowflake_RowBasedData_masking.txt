CREATE ROLE IF NOT EXISTS MANAGER_NORTH;
CREATE ROLE IF NOT EXISTS MANAGER_SOUTH;
CREATE ROLE IF NOT EXISTS MANAGER_EAST;
CREATE ROLE IF NOT EXISTS MANAGER_WEST;


grant role manager_north to user DUSHYANTSHARMA38;
grant role manager_south to user DUSHYANTSHARMA38;
grant role manager_east to user DUSHYANTSHARMA38;
grant role manager_west to user DUSHYANTSHARMA38;



grant usage on database  demo_db to role manager_south;
grant select on table demo_db.public.manager_data  to role manager_south;


grant usage on database demo_db to role manager_north;
grant select on table demo_db.public.manager_data to role manager_north;


grant usage on database demo_db to role manager_east;
grant select on table demo_db.public.manager_data to role manager_east;

grant usage on database demo_db to role manager_west;
grant select on table demo_db.public.manager_data to role manager_west;


CREATE OR REPLACE ROW ACCESS POLICY demo_db.public.direction_policy
AS (direction STRING)
RETURNS BOOLEAN ->
    CASE
        WHEN CURRENT_ROLE() = 'ACCOUNTADMIN' THEN TRUE
        WHEN CURRENT_ROLE() = 'MANAGER_NORTH' AND direction = 'NORTH' THEN TRUE
        WHEN CURRENT_ROLE() = 'MANAGER_SOUTH' AND direction = 'SOUTH' THEN TRUE
        WHEN CURRENT_ROLE() = 'MANAGER_EAST' AND direction = 'EAST' THEN TRUE
        WHEN CURRENT_ROLE() = 'MANAGER_WEST' AND direction = 'WEST' THEN TRUE
        ELSE FALSE
    END;


alter table demo_db.public.manager_data
add row access policy demo_db.public.direction_policy
on (direction);
