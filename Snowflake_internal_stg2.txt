CREATE OR REPLACE FILE FORMAT my_csv_format
    TYPE = CSV
    SKIP_HEADER = 1;

SHOW FILE FORMATS LIKE 'MY_CSV_FORMAT';




CREATE OR REPLACE TABLE student_data (
    id INT,
    name VARCHAR(50),
    age INT,
    name3 VARCHAR(50)
);

COPY INTO student_data (id, name, age, name3)
FROM (
    SELECT
        t.$1,
        t.$2,
        t.$3,
        SUBSTR(t.$2, 1, 3)
    FROM @internal_stg
        (FILE_FORMAT => 'MY_CSV_FORMAT') t
);

SELECT * FROM student_data;
