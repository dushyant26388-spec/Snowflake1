USE DATABASE TSESTDB;

CREATE SCHEMA IF NOT EXISTS BRONZE;
USE SCHEMA BRONZE;

CREATE STAGE CSV_STAGE
URL = 's3://pos-iot-bucket/CSV/'
CREDENTIALS = (
    AWS_KEY_ID = ''
    AWS_SECRET_KEY = ''
);

list @csv_stage;


CREATE STAGE JSON_STAGE
URL = 's3://pos-iot-bucket/JSON/'
CREDENTIALS = (
    AWS_KEY_ID = ''
    AWS_SECRET_KEY = ''
);

list @json_stage;


CREATE FILE FORMAT CSV_FORMAT
TYPE = 'CSV'
SKIP_HEADER = 1
FIELD_OPTIONALLY_ENCLOSED_BY = '"';

CREATE FILE FORMAT JSON_FORMAT
TYPE = 'JSON';

CREATE TABLE BRONZE_CSV (
    TRANSACTION_ID VARCHAR(50),
    STORE_ID VARCHAR(50),
    STORE_NAME VARCHAR(100),
    STORE_CITY VARCHAR(100),
    STORE_REGION VARCHAR(100),
    CASHIER_ID VARCHAR(50),
    CUSTOMER_ID VARCHAR(50),
    TRANSACTION_DATE DATE,
    TRANSACTION_TIME TIME,
    PRODUCT_SKU VARCHAR(50),
    PRODUCT_NAME VARCHAR(200),
    CATEGORY VARCHAR(100),
    SUBCATEGORY VARCHAR(100),
    QUANTITY NUMBER,
    UNIT_PRICE NUMBER(10,2),
    DISCOUNT_PCT NUMBER(5,2),
    TOTAL_AMOUNT NUMBER(10,2),
    PAYMENT_METHOD VARCHAR(50),
    LOYALTY_POINTS NUMBER
);

COPY INTO BRONZE_CSV
FROM @CSV_STAGE
FILE_FORMAT = (FORMAT_NAME = 'CSV_FORMAT')
ON_ERROR = 'ABORT_STATEMENT';

CREATE TABLE BRONZE_JSON (
    DATA VARIANT
);

COPY INTO BRONZE_JSON
FROM @JSON_STAGE
FILE_FORMAT = (FORMAT_NAME = 'JSON_FORMAT')
ON_ERROR = 'ABORT_STATEMENT';

SELECT * from bronze_json;

TYPE = 'JSON'
STRIP_OUTER_ARRAY = TRUE;

COPY INTO BRONZE_JSON

CREATE TABLE SILVER_JSON (
    EVENT_ID VARCHAR(100),
    DEVICE_ID VARCHAR(100),
    EVENT_TYPE VARCHAR(50),
    STORE_ID VARCHAR(50),
    STORE_NAME VARCHAR(100),
    EVENT_TIMESTAMP TIMESTAMP
);



CREATE STREAM BRONZE_JSON_STREAM
ON TABLE BRONZE_JSON
SHOW_INITIAL_ROWS = TRUE;


INSERT INTO SILVER_JSON
SELECT
    DATA:event_id::VARCHAR,
    DATA:device_id::VARCHAR,
    DATA:event_type::VARCHAR,
    DATA:store_id::VARCHAR,
    DATA:store_name::VARCHAR,
    DATA:timestamp::TIMESTAMP
FROM BRONZE_JSON_STREAM
WHERE METADATA$ACTION = 'INSERT';


SELECT * FROM SILVER_JSON;




-- create stream for silver cvs
CREATE STREAM BRONZE_CSV_STREAM
ON TABLE BRONZE_CSV
SHOW_INITIAL_ROWS = TRUE;

CREATE TABLE SILVER_CSV (
    TRANSACTION_ID VARCHAR(50),
    STORE_ID VARCHAR(50),
    STORE_NAME VARCHAR(100),
    STORE_CITY VARCHAR(100),
    TRANSACTION_DATE DATE,
    PRODUCT_SKU VARCHAR(50),
    PRODUCT_NAME VARCHAR(200),
    CATEGORY VARCHAR(100),
    QUANTITY NUMBER,
    UNIT_PRICE NUMBER(10,2),
    TOTAL_AMOUNT NUMBER(10,2),
    PAYMENT_METHOD VARCHAR(50)
);

INSERT INTO SILVER_CSV
SELECT
    TRANSACTION_ID,
    STORE_ID,
    STORE_NAME,
    STORE_CITY,
    TRANSACTION_DATE,
    PRODUCT_SKU,
    PRODUCT_NAME,
    CATEGORY,
    QUANTITY,
    UNIT_PRICE,
    TOTAL_AMOUNT,
    PAYMENT_METHOD
FROM BRONZE_CSV_STREAM
WHERE METADATA$ACTION = 'INSERT';
